# 第四章、对话历史

# 1、如何保留对话历史

在 LangChain 调用大模型中，默认是不保存历史对话的，换言之大模型不记得你上一次输入的信息

```python
# 第一次
response1 = model.invoke("我叫张三")

# 第二次（没传历史）
response2 = model.invoke("我叫什么？") # AI 不记得！
```

解决办法：每次调用必须传递完整的对话历史！

```python
from langchain.chat_models import init_chat_model
from dotenv import load_dotenv

load_dotenv(override=True)

llm = init_chat_model (
    model = "gpt-4o-mini",
    model_provider = "openai"
)

# 通过一个空 list 接受历史对话
conversation = []
# 第一次
conversation.append({"role": "user", "content": "我叫张三"})
response1 = llm.invoke(conversation)
# 关键：保存 AI 回复
conversation.append({"role": "assistant", "content": response1.content})
# 第二次（传递完整历史）
conversation.append({"role": "user", "content": "我叫什么？"})
response2 = llm.invoke(conversation) # AI 记得！
print(response2.content)
```

# 2、对话历史优化

问题：上述这样保留对话历史，对话历史会越来越长，消耗大量 tokens 和成本。

解决方法：只保留最近N轮对话

```python
from langchain.chat_models import init_chat_model
from dotenv import load_dotenv

def keep_message (messages, max_pairs=3):
    """
    保留最近N轮对话，默认3
    """
    # 分离 system 和对话
    system_msgs = [m for m in messages if m.get("role") == "system"]
    conversation_msgs = [m for m in messages if m.get("role") != "system"]
    
    # 只保留最近的
    recent_msgs = conversation_msgs[-(max_pairs * 2):]
    
    return system_msgs + recent_msgs

load_dotenv(override=True)

model = init_chat_model (
    model = "gpt-4o-mini",
    model_provider = "openai"
)

# 初始化
long_conversation = [
    {"role": "system", "content": "你是 Python 导师"}
]

# 第 1 轮
long_conversation.append({"role": "user", "content": "什么是列表？用一句解释"})
r1 = model.invoke(long_conversation)
long_conversation.append({"role": "assistant", "content": r1.content})

# 第 2 轮
long_conversation.append({"role": "user", "content": "列表和元组有什么区别？用一句解释"})
r2 = model.invoke(long_conversation)
long_conversation.append({"role": "assistant", "content": r2.content})

# 第 3 轮
long_conversation.append({"role": "user", "content": "什么是字典呢？用一句解释"})
r3 = model.invoke(long_conversation)
long_conversation.append({"role": "assistant", "content": r3.content})

print(f"原始消息数: {len(long_conversation)}")  # 7

# 优化：只保留最近 2 轮
optimized = keep_message(long_conversation, max_pairs=2)

print(f"优化后消息数: {len(optimized)}")  # 5
print(f"保留的内容: system + 最近2轮对话")

# 添加新的用户问题
optimized.append({"role": "user", "content": "我第一个问题问的是什么？"})
# 使用优化后的历史
response = model.invoke(optimized)
print(f"\nAI 回复: {response.content}")
```

但是生产环境一般不这样保留历史对话，而是保存到文件或者数据库

```python
import os
from pathlib import Path
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.runnables.history import RunnableWithMessageHistory
from langchain_community.chat_message_histories import FileChatMessageHistory
from langchain_core.messages import trim_messages, SystemMessage, HumanMessage
from dotenv import load_dotenv

# 1. 基础配置
load_dotenv(override=True)
HISTORY_DIR = Path("./chat_histories")
HISTORY_DIR.mkdir(exist_ok=True)

def get_file_session_history(session_id: str):
    file_path = HISTORY_DIR / f"{session_id}.json"
    return FileChatMessageHistory(str(file_path))

# 2. 定义裁剪器（Trimmer）
# 这里配置为：只保留最近的 4 条消息（即约 2 轮对话），并确保消息切分完整
trimmer = trim_messages(
    max_tokens=4,           # 限制消息数量或 Token 数
    token_counter=len,      # 使用消息条数计数器 (也可传入模型来做真实 Token 计算)
    strategy="last",        # 只保留最新的消息 (滑动窗口)
    start_on="human",       # 截断后确保第一条消息永远是 HumanMessage，避免格式错误
    include_system=False,   # 裁剪时不包含系统提示词
)

# 3. 创建 Prompt 和 LLM
prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一个友好的 AI 助手。"),
    MessagesPlaceholder(variable_name="history"), # 注意：这里填入的是裁剪后的历史
    ("human", "{input}"),
])

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.7)

# 4. 构建包含“消息裁剪”能力的 Chain
# 关键点：在传入 prompt 之前，先通过 trimmer 对 history 进行过滤
chain = (
    {
        "input": lambda x: x["input"],
        # 从原始 history 中提取并进行裁剪
        "history": lambda x: trimmer.invoke(x["history"]),
    }
    | prompt
    | llm
)

# 5. 用 RunnableWithMessageHistory 包装
conversation_chain = RunnableWithMessageHistory(
    chain,
    get_file_session_history,
    input_messages_key="input",
    history_messages_key="history",
)

# 6. 测试剪裁效果
if __name__ == "__main__":
    config = {"configurable": {"session_id": "test_trim_user"}}

    # 连续发起 4 轮对话
    conversation_chain.invoke({"input": "我的幸运数字是 7"}, config=config)
    conversation_chain.invoke({"input": "我喜欢吃苹果"}, config=config)
    conversation_chain.invoke({"input": "今天天气真不错"}, config=config)
    
    # 第 4 轮询问（此时“幸运数字是 7”已经被划出 4 条消息之外，AI 将不再记得）
    res = conversation_chain.invoke({"input": "你还记得我的幸运数字是多少吗？"}, config=config)
    print("AI 回答:", res.content)
```

# 项目实战：多轮聊天机器人

```python
from langchain.chat_models import init_chat_model
from dotenv import load_dotenv
from langchain_core.messages import SystemMessage, HumanMessage, AIMessage
import os

def keep_message (messages, max_pairs=3):
    """
    保留最近N轮对话，默认3
    """
    # 分离 system 和对话
    system_msgs = [m for m in messages if isinstance(m, SystemMessage)]
    conversation_msgs = [m for m in messages if not isinstance(m, SystemMessage)]
    
    # 只保留最近的
    recent_msgs = conversation_msgs[-(max_pairs * 2):]
    
    return system_msgs + recent_msgs

    
MODEL_NAME = "gpt-4o-mini"
MODEL_PROVIDER = "openai"
MAX_MESSAGES_HISTORY = 10
EXIT_WORD = "quit"

load_dotenv(override=True)

model = init_chat_model(
    model = MODEL_NAME,
    model_provider = MODEL_PROVIDER
)

messages = [
    SystemMessage("你是一个友善的AI助手，回答用户的问题。")
]

print("请输入您的问题，我尽量帮您解答。当输入{}的时候结束对话".format(EXIT_WORD))

i = 1
while True :
    print("第 {} 轮对话开始：".format(i))
    user_input = input("请输入：")
    if user_input == EXIT_WORD:
        print("对话已结束，很高兴为您服务")
        break
    messages.append(HumanMessage(user_input))
    print("AI助手回复：", end="", flush=True)
    reply_content = ""
    memory_messages = keep_message(messages, max_pairs=MAX_MESSAGES_HISTORY)
    for chunk in model.stream(memory_messages):
        if chunk.content:
            print(chunk.content, end="", flush=True)
            reply_content += chunk.content
    print("\n第 {} 轮对话结束！".format(i))
    i+=1
    messages.append(AIMessage(reply_content))
```
